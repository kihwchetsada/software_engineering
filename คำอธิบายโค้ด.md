<?php
// ส่วนของ PHP ที่ใช้ประมวลผลหลังจากกดปุ่ม 'Calculate'

// ข้อมูลกิจกรรมและ dependencies
$activities_data = [
    'A' => ['dependencies' => ['']],
    'B' => ['dependencies' => ['A']],
    // ฯลฯ
];

// สร้างคลาส Activity สำหรับจัดการค่าของแต่ละกิจกรรม
class Activity {
    // กำหนดคุณสมบัติเริ่มต้น เช่น a, m, b, t, variance ฯลฯ
    public $name, $a, $m, $b, $t, $variance, $es, $ef, $ls, $lf, $slack, $dependencies;

    // ฟังก์ชันคอนสตรัคเตอร์เพื่อกำหนดค่าเริ่มต้น
    public function __construct($name, $a, $m, $b, $dependencies) {
        $this->name = $name;
        $this->a = $a;
        $this->m = $m;
        $this->b = $b;
        $this->t = ($a + 4 * $m + $b) / 6;  // คำนวณค่า t (เวลาที่คาดหวัง)
        $this->variance = pow(($b - $a) / 6, 2);  // คำนวณค่า variance (σ²)
        $this->dependencies = $dependencies;  // กิจกรรมที่ต้องทำก่อน
    }
}

// ตัวแปรสำหรับเก็บกิจกรรมทั้งหมด
$activities = [];
$showResults = false;

// ตรวจสอบว่าฟอร์มถูกส่งมาหรือไม่
if ($_SERVER['REQUEST_METHOD'] == 'POST' && isset($_POST['calculate'])) {
    // รับค่าจากฟอร์มและสร้างออบเจกต์ Activity แต่ละตัว
    foreach ($activities_data as $name => $data) {
        $a = $_POST["a_$name"] ?? 0;
        $m = $_POST["m_$name"] ?? 0;
        $b = $_POST["b_$name"] ?? 0;
        $activities[$name] = new Activity($name, $a, $m, $b, $data['dependencies']);
    }

    // คำนวณค่า ES และ EF (Early Start, Early Finish)
    calculateESandEF($activities);

    // คำนวณค่า LS และ LF (Late Start, Late Finish)
    $activities = calculateLSandLF($activities);

    // คำนวณค่า Slack (เวลาว่าง)
    $activities = calculateSlack($activities);

    // หาเส้นทางวิกฤต (Critical Path)
    $criticalPath = findCriticalPath($activities);

    // แสดงผลลัพธ์
    $showResults = true;
}

// ฟังก์ชันคำนวณค่า ES และ EF
function calculateESandEF($activities) {
    foreach ($activities as $activity) {
        if (empty($activity->dependencies[0])) {
            // กำหนดค่า ES และ EF ให้กับกิจกรรมที่ไม่มี dependencies
            $activity->es = 0;
            $activity->ef = $activity->t;
        } else {
            // คำนวณค่า ES และ EF จากกิจกรรมที่เป็น dependencies
            $activity->es = max(array_map(function($dep) use ($activities) {
                return $activities[$dep]->ef;
            }, $activity->dependencies));
            $activity->ef = $activity->es + $activity->t;
        }
    }
}

// ฟังก์ชันคำนวณค่า LS และ LF
function calculateLSandLF($activities) {
    $maxEF = max(array_map(function($activity) {
        return $activity->ef;
    }, $activities));

    foreach (array_reverse($activities) as $activity) {
        if (empty($activity->dependencies[0])) {
            // กำหนดค่า LS และ LF ให้กับกิจกรรมที่ไม่มี dependencies
            $activity->lf = $maxEF;
            $activity->ls = $activity->lf - $activity->t;
        } else {
            // คำนวณค่า LS และ LF จากกิจกรรมที่เป็น dependencies
            $activity->lf = min(array_map(function($dep) use ($activities) {
                return $activities[$dep]->ls;
            }, $activity->dependencies));
            $activity->ls = $activity->lf - $activity->t;
        }
    }
    return $activities;
}

// ฟังก์ชันคำนวณค่า Slack
function calculateSlack($activities) {
    foreach ($activities as $activity) {
        // Slack คือส่วนต่างระหว่าง LS และ ES
        $activity->slack = $activity->ls - $activity->es;
    }
    return $activities;
}

// ฟังก์ชันหา Critical Path (เส้นทางวิกฤต)
function findCriticalPath($activities) {
    // เส้นทางวิกฤตจะประกอบด้วยกิจกรรมที่มีค่า Slack = 0
    return array_keys(array_filter($activities, function($activity) {
        return $activity->slack == 0;
    }));
}

    $T = max(array_map(function($activity) {

    }, $activities));

    // กำหนดค่า X ตามที่ผู้ใช้ใส่ หรือคำนวณจาก Critical Path

        $X = floatval($_POST['custom_x']);
    } else {
        $X = array_sum(array_map(function($activity) {
            return ($activity->slack == 0) ? $activity->t : 0;
        }, $activities));
    }

    // คำนวณค่า Variance ของเส้นทางวิกฤต
    $variance_sum = array_sum(array_map(function($activity) {
        return ($activity->slack == 0) ? $activity->variance : 0;
    }, $activities));
    ?>

    <p><strong>Variance Sum (σ²):</strong> <?= $variance_sum ?> </p>
